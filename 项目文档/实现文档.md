#### 2.1.2 控制器实现

```java
// UserController.java
@Controller
@RequestMapping("/user")
public class UserController {
    
    @Autowired
    private UserServiceImpl userService;
    
    @Autowired
    private RedisTemplate redisTemplate;
    
    // 用户注册
    @RequestMapping("/register")
    @ResponseBody
    public Result register(@RequestParam String cellPhone, @RequestParam String authCode, HttpSession session) {
        // 验证码校验
        if(authCode.equals(redisTemplate.opsForValue().get(cellPhone + "authCode"))) {
            // 验证成功后删除验证码
            redisTemplate.delete(cellPhone + "authCode");
            
            User user = new User();
            user.setUserName(cellPhone);
            user.setCellPhone(cellPhone);
            user.setUserAvatar("user.webp");
            
            // 检查用户是否已存在
            User existingUser = userService.selectUserByCellPhone(cellPhone);
            if(existingUser == null) {
                // 新用户注册
                int result = userService.insertUser(user);
                if(result > 0) {
                    User newUser = userService.selectUserByCellPhone(cellPhone);
                    session.setAttribute("user", newUser);
                    
                    // 初始化每日打卡状态
                    userService.insertEverydayStatusById(newUser.getId());
                    EverydayStatus status = userService.selectEverydayStatusById(newUser.getId());
                    session.setAttribute("everydayStatus", status);
                    
                    return Result.success();
                }
            } else {
                // 已存在用户直接登录
                if(existingUser.getStatus() == 1) {
                    return new Result(3, existingUser.getBanDay()+"");
                }
                session.setAttribute("user", existingUser);
                EverydayStatus status = userService.selectEverydayStatusById(existingUser.getId());
                session.setAttribute("everydayStatus", status);
                return Result.success();
            }
        }
        return Result.failure("验证码错误或已过期");
    }
    
    // 用户登录
    @RequestMapping("/login")
    @ResponseBody
    public Result login(@RequestParam String cellPhone, @RequestParam String userPassword, HttpSession session) {
        User user = userService.selectUserByCellPhone(cellPhone);
        if(user == null) {
            return new Result(1, "用户不存在");
        }
        
        if(user.getUserPassword().equals(userPassword)) {
            if(user.getStatus() == 1) {
                return new Result(3, user.getBanDay()+"");
            }
            session.setAttribute("user", user);
            EverydayStatus status = userService.selectEverydayStatusById(user.getId());
            session.setAttribute("everydayStatus", status);
            return Result.success();
        }
        return new Result(4, "密码错误");
    }
}
```

```java
@Service
public class StudyStatusServiceImpl implements StudyStatusService {
    
    @Autowired
    private StudyStatusMapper studyStatusMapper;
    
    // 点赞功能
    public Result likeStudyStatus(int userId, int studyStatusId) {
        try {
            // 检查是否已点赞
            LikeRecord existingLike = studyStatusMapper.selectLikeRecord(userId, studyStatusId);
            if(existingLike != null) {
                // 取消点赞
                studyStatusMapper.deleteLikeRecord(userId, studyStatusId);
                studyStatusMapper.decreaseLikeCount(studyStatusId);
                return Result.success("取消点赞成功");
            } else {
                // 添加点赞
                LikeRecord likeRecord = new LikeRecord();
                likeRecord.setUserId(userId);
                likeRecord.setStudyStatusId(studyStatusId);
                likeRecord.setCreateTime(new Date());
                
                studyStatusMapper.insertLikeRecord(likeRecord);
                studyStatusMapper.increaseLikeCount(studyStatusId);
                return Result.success("点赞成功");
            }
        } catch (Exception e) {
            return Result.failure("操作失败");
        }
    }
```

```java
// WebSocketConfig.java
@Configuration
@EnableWebSocket
public class WebSocketConfig implements WebSocketConfigurer {
    
    @Override
    public void registerWebSocketHandlers(WebSocketHandlerRegistry registry) {
        registry.addHandler(new StudyRoomWebSocketHandler(), "/websocket/{roomId}")
                .setAllowedOrigins("*");
    }
}

// StudyRoomWebSocketHandler.java
@Component
public class StudyRoomWebSocketHandler extends TextWebSocketHandler {
    
    private static final Map<String, Set<WebSocketSession>> roomSessions = new ConcurrentHashMap<>();
    
    @Override
    public void afterConnectionEstablished(WebSocketSession session) throws Exception {
        String roomId = getRoomIdFromSession(session);
        
        // 添加到房间会话集合
        roomSessions.computeIfAbsent(roomId, k -> ConcurrentHashMap.newKeySet())
                   .add(session);
        
        // 发送用户加入消息
        broadcastMessage(roomId, createJoinMessage(session));
    }
    
    @Override
    protected void handleTextMessage(WebSocketSession session, TextMessage message) throws Exception {
        String roomId = getRoomIdFromSession(session);
        
        // 处理消息并广播
        String processedMessage = processMessage(message.getPayload(), session);
        broadcastMessage(roomId, processedMessage);
    }
    
    private void broadcastMessage(String roomId, String message) {
        Set<WebSocketSession> sessions = roomSessions.get(roomId);
        if(sessions != null) {
            sessions.forEach(session -> {
                try {
                    if(session.isOpen()) {
                        session.sendMessage(new TextMessage(message));
                    }
                } catch (IOException e) {
                    e.printStackTrace();
                }
            });
        }
    }
}
```

### 2.4 缓存实现

#### 2.4.1 Redis配置

```java
// RedisConfig.java
@Configuration
public class RedisConfig {
    
    @Bean
    public RedisTemplate<String, Object> redisTemplate(RedisConnectionFactory factory) {
        RedisTemplate<String, Object> template = new RedisTemplate<>();
        template.setConnectionFactory(factory);
        
        // 使用Jackson序列化
        Jackson2JsonRedisSerializer<Object> serializer = new Jackson2JsonRedisSerializer<>(Object.class);
        ObjectMapper mapper = new ObjectMapper();
        mapper.setVisibility(PropertyAccessor.ALL, JsonAutoDetect.Visibility.ANY);
        mapper.activateDefaultTyping(LazyCollectionTypeResolver.instance, ObjectMapper.DefaultTyping.NON_FINAL);
        serializer.setObjectMapper(mapper);
        
        template.setKeySerializer(new StringRedisSerializer());
        template.setValueSerializer(serializer);
        template.setHashKeySerializer(new StringRedisSerializer());
        template.setHashValueSerializer(serializer);
        template.afterPropertiesSet();
        
        return template;
    }
}
```

#### 2.4.2 缓存服务实现

```java
// RedisService.java
@Service
public class RedisService {
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    // 用户信息缓存
    public void cacheUserInfo(User user) {
        String key = "user:info:" + user.getId();
        redisTemplate.opsForValue().set(key, user, 30, TimeUnit.MINUTES);
    }
    
    public User getCachedUserInfo(int userId) {
        String key = "user:info:" + userId;
        return (User) redisTemplate.opsForValue().get(key);
    }
    
    // 动态热度排名
    public void updateStudyStatusHotScore(int statusId, double score) {
        String key = "hot:study:status";
        redisTemplate.opsForZSet().add(key, String.valueOf(statusId), score);
    }
    
    public Set<Object> getHotStudyStatus(int count) {
        String key = "hot:study:status";
        return redisTemplate.opsForZSet().reverseRange(key, 0, count - 1);
    }
}
```

```java
// GlobalExceptionHandler.java
@ControllerAdvice
public class GlobalExceptionHandler {
    
    private static final Logger logger = LoggerFactory.getLogger(GlobalExceptionHandler.class);
    
    /**
     * 处理所有不可知的异常
     */
    @ExceptionHandler(Exception.class)
    @ResponseBody
    public Result handleException(Exception e) {
        logger.error("系统异常: ", e);
        return Result.failure("系统异常，请稍后重试");
    }
    
    /**
     * 处理业务异常
     */
    @ExceptionHandler(BusinessException.class)
    @ResponseBody
    public Result handleBusinessException(BusinessException e) {
        logger.warn("业务异常: {}", e.getMessage());
        return Result.failure(e.getMessage());
    }
}
```
